using System;
using System.Collections.Generic;
using QSL.Abstractions.Base;
using QSL.Core.Provenance;
using QSL.Core.Runtime;

namespace QSL.Core.Base
{
    public abstract class QslBaseObject : IQslObject
    {
        public Guid Id { get; } = Guid.NewGuid();

        private string? _slug;

        public string Slug => _slug ??= Id.ToSlug();

        /// <summary>
        /// Extends this object's existing heritage with another
        /// processing or transformation identifier.
        /// </summary>
        public void AppendSlug(string suffix)
        {
            _slug = $"{Slug}-{suffix}";
        }

        /// <summary>
        /// Establishes inherited or reconstructed heritage without
        /// requiring this object to have experienced each represented
        /// transformation itself.
        ///
        /// Intended for controlled operations such as cloning,
        /// deep copying, or restoration.
        /// </summary>
        protected void SetSlug(string newSlug)
        {
            _slug = newSlug;
        }

        private readonly List<QslProvenanceEntry> _provenance = new();

        public IReadOnlyList<QslProvenanceEntry> Provenance => _provenance;

        protected QslProvenanceEntry AddProvenance(
            QslProvenanceType type,
            QslProvenanceCategory category,
            string source,
            string message,
            Dictionary<string, object>? metadata = null)
        {
            var metadataSnapshot = metadata is null
                ? new Dictionary<string, object>()
                : new Dictionary<string, object>(metadata);

            var entry = new QslProvenanceEntry
            {
                SubjectId = Id,
                SubjectSlug = Slug,
                EntryType = type,
                Category = category,
                Source = source,
                Message = message,
                Metadata = metadataSnapshot
            };

            _provenance.Add(entry);

            QslRuntime.Instance.ProvenanceBus.Publish(entry);

            return entry;
        }

        public abstract string Type { get; }
    }
}

        public static QslTokenAttribute Info(
            this QslTokenKind kind)
        {
            if (Metadata.TryGetValue(kind, out var info))
                return info;

            throw new InvalidOperationException(
                $"No metadata is registered for {kind}.");
        }

        public static int Precedence(
            this QslTokenKind kind)
            => kind.Info().Precedence;

        public static int UnaryPrecedence(
            this QslTokenKind kind)
            => kind.Info().UnaryPrecedence;

        public static bool RightAssociative(
            this QslTokenKind kind)
            => kind.Info().RightAssociative;

        public static bool IsUnary(
            this QslTokenKind kind)
            => kind.Info().Unary;

        public static bool IsBinary(
            this QslTokenKind kind)
            => kind.Info().Binary;

        public static string Symbol(
            this QslTokenKind kind)
            => kind.Info().Symbol;

        private static IReadOnlyDictionary<
            QslTokenKind,
            QslTokenAttribute> BuildMetadata()
        {
            var metadata =
                new Dictionary<QslTokenKind, QslTokenAttribute>();

            foreach (QslTokenKind kind in Enum.GetValues(typeof(QslTokenKind)))
            {
                var member = typeof(QslTokenKind)
                    .GetMember(kind.ToString());

                if (member.Length == 0)
                {
                    throw new InvalidOperationException(
                        $"Unable to reflect QSL token kind {kind}.");
                }

                var attribute =
                    member[0].GetCustomAttribute<QslTokenAttribute>();

                if (attribute is null)
                {
                    throw new InvalidOperationException(
                        $"QslTokenKind.{kind} has no " +
                        $"{nameof(QslTokenAttribute)}.");
                }

                if (attribute.Unary &&
                    attribute.UnaryPrecedence <= 0)
                {
                    throw new InvalidOperationException(
                        $"Unary token QslTokenKind.{kind} must " +
                        "define UnaryPrecedence.");
                }

                if (attribute.Binary &&
                    attribute.Precedence <= 0)
                {
                    throw new InvalidOperationException(
                        $"Binary token QslTokenKind.{kind} must " +
                        "define Precedence.");
                }

                metadata.Add(kind, attribute);
            }

            return metadata;
        }

        public static Type ExecutorType(
            this QslTokenKind kind)
        {
            return kind.Info().ExecutorType
                   ?? throw new InvalidOperationException(
                       $"QslTokenKind.{kind} has no executor.");
        }

        public static QslEvaluationStrategy EvaluationStrategy(
            this QslTokenKind kind)
            => kind.Info().EvaluationStrategy;
    }

using System;
using System.Collections.Generic;
using System.Text;

namespace QSL.Core.Lexer
{
    public sealed class QslTokenizer
    {
        private readonly string _source;
        private int _pos;

        public QslTokenizer(string source)
        {
            _source = source ?? string.Empty;
            _pos = 0;
        }

        public IEnumerable<QslToken> Tokenize()
        {
            while (!IsAtEnd())
            {
                var current = Peek();

                if (char.IsWhiteSpace(current))
                {
                    Advance();
                    continue;
                }

                if (char.IsLetter(current) || current == '_')
                {
                    yield return ReadIdentifier();
                    continue;
                }

                if (char.IsDigit(current))
                {
                    yield return ReadNumber();
                    continue;
                }

                switch (current)
                {
                    case '"':
                    {
                        foreach (var token in ReadStringDouble())
                            yield return token;

                        continue;
                    }

                    case '\'':
                    {
                        foreach (var token in ReadStringSingle())
                            yield return token;

                        continue;
                    }

                    case '$':
                        yield return MakeToken(
                            QslTokenKind.Dollar,
                            "$");

                        Advance();
                        continue;

                    case '(':
                        yield return MakeToken(
                            QslTokenKind.LeftParen,
                            "(");

                        Advance();
                        continue;

                    case ')':
                        yield return MakeToken(
                            QslTokenKind.RightParen,
                            ")");

                        Advance();
                        continue;

                    case '{':
                        yield return MakeToken(
                            QslTokenKind.LeftBrace,
                            "{");

                        Advance();
                        continue;

                    case '}':
                        yield return MakeToken(
                            QslTokenKind.RightBrace,
                            "}");

                        Advance();
                        continue;

                    case ',':
                        yield return MakeToken(
                            QslTokenKind.Comma,
                            ",");

                        Advance();
                        continue;

                    case ';':
                        yield return MakeToken(
                            QslTokenKind.Semicolon,
                            ";");

                        Advance();
                        continue;

                    case '+':
                        yield return MakeToken(
                            QslTokenKind.Plus,
                            "+");

                        Advance();
                        continue;

                    case '-':
                        yield return MakeToken(
                            QslTokenKind.Minus,
                            "-");

                        Advance();
                        continue;

                    case '*':
                        yield return MakeToken(
                            QslTokenKind.Star,
                            "*");

                        Advance();
                        continue;

                    case '/':
                        yield return MakeToken(
                            QslTokenKind.Slash,
                            "/");

                        Advance();
                        continue;

                    case '%':
                        yield return MakeToken(
                            QslTokenKind.Percent,
                            "%");

                        Advance();
                        continue;

                    case '&':
                        if (MatchNext('&'))
                        {
                            yield return MakeToken(
                                QslTokenKind.AndAnd,
                                "&&");

                            continue;
                        }

                        yield return MakeUnknownToken(current);
                        Advance();
                        continue;

                    case '|':
                        if (MatchNext('|'))
                        {
                            yield return MakeToken(
                                QslTokenKind.OrOr,
                                "||");

                            continue;
                        }

                        yield return MakeUnknownToken(current);
                        Advance();
                        continue;

                    case '=':
                        if (MatchNext('='))
                        {
                            yield return MakeToken(
                                QslTokenKind.EqualEqual,
                                "==");

                            continue;
                        }

                        if (MatchNext('>'))
                        {
                            yield return MakeToken(
                                QslTokenKind.Arrow,
                                "=>");

                            continue;
                        }

                        yield return MakeToken(
                            QslTokenKind.Equal,
                            "=");

                        Advance();
                        continue;

                    case '!':
                        if (MatchNext('='))
                        {
                            yield return MakeToken(
                                QslTokenKind.BangEqual,
                                "!=");

                            continue;
                        }

                        yield return MakeToken(
                            QslTokenKind.Bang,
                            "!");

                        Advance();
                        continue;

                    case '<':
                        if (MatchNext('='))
                        {
                            yield return MakeToken(
                                QslTokenKind.LessEqual,
                                "<=");

                            continue;
                        }

                        yield return MakeToken(
                            QslTokenKind.Less,
                            "<");

                        Advance();
                        continue;

                    case '>':
                        if (MatchNext('='))
                        {
                            yield return MakeToken(
                                QslTokenKind.GreaterEqual,
                                ">=");

                            continue;
                        }

                        yield return MakeToken(
                            QslTokenKind.Greater,
                            ">");

                        Advance();
                        continue;

                    case '?':
                        if (MatchNext('?'))
                        {
                            yield return MakeToken(
                                QslTokenKind.QuestionQuestion,
                                "??");

                            continue;
                        }

                        yield return MakeToken(
                            QslTokenKind.Question,
                            "?");

                        Advance();
                        continue;

                    case ':':
                        yield return MakeToken(
                            QslTokenKind.Colon,
                            ":");

                        Advance();
                        continue;

                    default:
                        yield return MakeUnknownToken(current);
                        Advance();
                        continue;
                }
            }

            yield return MakeToken(
                QslTokenKind.EndOfFile,
                string.Empty);
        }

        private bool IsAtEnd()
        {
            return _pos >= _source.Length;
        }

        private char Peek()
        {
            return IsAtEnd()
                ? '\0'
                : _source[_pos];
        }

        private char Advance()
        {
            return IsAtEnd()
                ? '\0'
                : _source[_pos++];
        }

        /// <summary>
        /// Checks whether the character immediately after the current
        /// position matches the expected character.
        ///
        /// When matched, both the current character and the following
        /// character are consumed.
        /// </summary>
        private bool MatchNext(char expected)
        {
            var nextPosition = _pos + 1;

            if (nextPosition >= _source.Length)
                return false;

            if (_source[nextPosition] != expected)
                return false;

            _pos += 2;
            return true;
        }

        private QslToken MakeToken(
            QslTokenKind kind,
            string text)
        {
            return new QslToken(kind, text);
        }

        private QslToken MakeUnknownToken(char value)
        {
            return MakeToken(
                QslTokenKind.Unknown,
                value.ToString());
        }

        private QslToken ReadIdentifier()
        {
            var start = _pos;

            while (!IsAtEnd() &&
                   (char.IsLetterOrDigit(Peek()) ||
                    Peek() == '_'))
            {
                Advance();
            }

            var text = _source.Substring(
                start,
                _pos - start);

            switch (text)
            {
                case "true":
                    return MakeToken(
                        QslTokenKind.True,
                        text);

                case "false":
                    return MakeToken(
                        QslTokenKind.False,
                        text);

                default:
                    return MakeToken(
                        QslTokenKind.Identifier,
                        text);
            }
        }

        private QslToken ReadNumber()
        {
            var start = _pos;

            while (!IsAtEnd() &&
                   char.IsDigit(Peek()))
            {
                Advance();
            }

            if (!IsAtEnd() &&
                Peek() == '.' &&
                HasNextDigit())
            {
                Advance();

                while (!IsAtEnd() &&
                       char.IsDigit(Peek()))
                {
                    Advance();
                }
            }

            var text = _source.Substring(
                start,
                _pos - start);

            return MakeToken(
                QslTokenKind.Number,
                text);
        }

        private bool HasNextDigit()
        {
            var nextPosition = _pos + 1;

            return nextPosition < _source.Length &&
                   char.IsDigit(_source[nextPosition]);
        }

        private IEnumerable<QslToken> ReadStringDouble()
        {
            Advance();

            yield return MakeToken(
                QslTokenKind.StringStartDouble,
                "\"");

            var content = new StringBuilder();

            while (!IsAtEnd())
            {
                var current = Advance();

                if (current == '"')
                {
                    yield return MakeToken(
                        QslTokenKind.StringContent,
                        content.ToString());

                    yield return MakeToken(
                        QslTokenKind.StringEndDouble,
                        "\"");

                    yield break;
                }

                if (current == '\\')
                {
                    if (IsAtEnd())
                    {
                        throw new InvalidOperationException(
                            "Unterminated escape sequence in " +
                            "double-quoted string.");
                    }

                    var escaped = Advance();

                    content.Append(
                        ResolveEscapeCharacter(escaped));

                    continue;
                }

                content.Append(current);
            }

            throw new InvalidOperationException(
                "Unterminated double-quoted string.");
        }

        private IEnumerable<QslToken> ReadStringSingle()
        {
            Advance();

            yield return MakeToken(
                QslTokenKind.StringStartSingle,
                "'");

            var content = new StringBuilder();

            while (!IsAtEnd())
            {
                var current = Advance();

                if (current == '\'')
                {
                    yield return MakeToken(
                        QslTokenKind.StringContent,
                        content.ToString());

                    yield return MakeToken(
                        QslTokenKind.StringEndSingle,
                        "'");

                    yield break;
                }

                content.Append(current);
            }

            throw new InvalidOperationException(
                "Unterminated single-quoted string.");
        }

        private char ResolveEscapeCharacter(char escaped)
        {
            switch (escaped)
            {
                case '"':
                    return '"';

                case '\'':
                    return '\'';

                case '\\':
                    return '\\';

                case 'n':
                    return '\n';

                case 'r':
                    return '\r';

                case 't':
                    return '\t';

                case '$':
                    return '$';

                case ')':
                    return ')';

                default:
                    return escaped;
            }
        }
    }

using System.Collections.Generic;
using System.Globalization;
using QSL.Abstractions.Runtime;
using QSL.Core.Ast;
using QSL.Core.Ast.Types;
using QSL.Core.Lexer;

namespace QSL.Core.Parser
{
    public sealed class QslParser
    {
        private readonly List<QslToken> _tokens;
        private readonly List<string> _errors = new List<string>();

        private int _pos;

        public IReadOnlyList<string> Errors => _errors;

        public QslParser(IEnumerable<QslToken> tokens)
        {
            _tokens = new List<QslToken>(tokens);
            _pos = 0;
        }

        public IQslNode ParseExpression()
        {
            var expression = ParseBinaryExpression(0);

            // Ternary:
            // condition ? trueExpression : falseExpression
            if (Peek().Kind == QslTokenKind.Question)
            {
                Advance();

                var trueExpression = ParseExpression();

                if (Peek().Kind != QslTokenKind.Colon)
                {
                    const string message =
                        "Expected ':' in ternary expression.";

                    _errors.Add(message);

                    return new QslErrorNode(message);
                }

                Advance();

                var falseExpression = ParseExpression();

                expression = new QslTernaryNode(
                    expression,
                    trueExpression,
                    falseExpression);
            }

            return expression;
        }

        private IQslNode ParseBinaryExpression(
            int minimumPrecedence)
        {
            var left = ParsePrefixExpression();

            while (true)
            {
                var operatorKind = Peek().Kind;

                if (!QslKindExtensions.IsBinary(operatorKind))
                    break;

                var precedence =
                    QslKindExtensions.Precedence(operatorKind);

                if (precedence < minimumPrecedence)
                    break;

                var rightAssociative =
                    QslKindExtensions.RightAssociative(
                        operatorKind);

                Advance();

                var nextMinimumPrecedence =
                    rightAssociative
                        ? precedence
                        : precedence + 1;

                var right = ParseBinaryExpression(
                    nextMinimumPrecedence);

                left = new QslBinaryNode(
                    operatorKind,
                    left,
                    right);
            }

            return left;
        }

        private IQslNode ParsePrefixExpression()
        {
            var operatorKind = Peek().Kind;

            if (!QslKindExtensions.IsUnary(operatorKind))
                return ParsePrimary();

            Advance();

            var unaryPrecedence =
                QslKindExtensions.UnaryPrecedence(
                    operatorKind);

            var operand = ParseBinaryExpression(
                unaryPrecedence);

            return new QslUnaryNode(
                operatorKind,
                operand);
        }

        private IQslNode ParsePrimary()
        {
            var token = Peek();

            switch (token.Kind)
            {
                case QslTokenKind.Number:
                    return ParseNumberLiteral();

                case QslTokenKind.Identifier:
                    return ParseIdentifierOrCall();

                case QslTokenKind.LeftParen:
                    return ParseGroupedExpression();

                case QslTokenKind.LeftBrace:
                    return ParseBlock();

                case QslTokenKind.Dollar:
                    return ParseEmbeddedExpression();

                case QslTokenKind.StringStartDouble:
                    return ParseStringLiteralDouble();

                case QslTokenKind.StringStartSingle:
                    return ParseStringLiteralSingle();

                case QslTokenKind.True:
                case QslTokenKind.False:
                    return ParseBooleanLiteral();

                case QslTokenKind.EndOfFile:
                {
                    const string message =
                        "Unexpected end of QSL expression.";

                    _errors.Add(message);

                    return new QslErrorNode(message);
                }

                default:
                {
                    var message =
                        "Unexpected token: " +
                        $"{token.Kind} '{token.Text}'.";

                    _errors.Add(message);
                    Advance();

                    return new QslErrorNode(message);
                }
            }
        }

        private IQslNode ParseNumberLiteral()
        {
            var token = Advance();

            double number;

            if (double.TryParse(
                    token.Text,
                    NumberStyles.Float,
                    CultureInfo.InvariantCulture,
                    out number))
            {
                return new QslLiteralNode(
                    QslLiteralKind.Number,
                    number);
            }

            var message =
                $"Invalid number literal: '{token.Text}'.";

            _errors.Add(message);

            return new QslErrorNode(message);
        }

        private IQslNode ParseIdentifierOrCall()
        {
            var identifierToken = Consume(
                QslTokenKind.Identifier,
                "Expected identifier.");

            var identifier =
                new QslIdentifierNode(
                    identifierToken.Text);

            if (Peek().Kind == QslTokenKind.LeftParen)
            {
                return ParseCallExpression(identifier);
            }

            return identifier;
        }

        private IQslNode ParseCallExpression(
            QslIdentifierNode target)
        {
            Consume(
                QslTokenKind.LeftParen,
                "Expected '(' after function name.");

            var arguments = new List<IQslNode>();

            if (Peek().Kind != QslTokenKind.RightParen)
            {
                while (Peek().Kind != QslTokenKind.EndOfFile)
                {
                    arguments.Add(ParseExpression());

                    if (Peek().Kind == QslTokenKind.RightParen)
                        break;

                    if (Peek().Kind != QslTokenKind.Comma)
                    {
                        var token = Peek();

                        var message =
                            "Expected ',' between arguments. " +
                            $"Found {token.Kind} '{token.Text}'.";

                        _errors.Add(message);

                        SynchronizeCallArguments();

                        if (Peek().Kind == QslTokenKind.RightParen)
                            break;

                        if (Peek().Kind == QslTokenKind.EndOfFile)
                            break;
                    }

                    if (Peek().Kind == QslTokenKind.Comma)
                        Advance();
                }
            }

            Consume(
                QslTokenKind.RightParen,
                "Expected ')' after arguments.");

            return new QslCallNode(
                target,
                arguments);
        }

        private IQslNode ParseGroupedExpression()
        {
            Consume(
                QslTokenKind.LeftParen,
                "Expected '('.");

            var expression = ParseExpression();

            Consume(
                QslTokenKind.RightParen,
                "Expected ')' after grouped expression.");

            return expression;
        }

        private IQslNode ParseEmbeddedExpression()
        {
            Consume(
                QslTokenKind.Dollar,
                "Expected '$' for embedded expression.");

            Consume(
                QslTokenKind.LeftParen,
                "Expected '(' after '$'.");

            var innerExpression = ParseExpression();

            Consume(
                QslTokenKind.RightParen,
                "Expected ')' to close embedded expression.");

            return new QslEmbeddedNode(
                innerExpression);
        }

        private IQslNode ParseStringLiteralDouble()
        {
            Consume(
                QslTokenKind.StringStartDouble,
                "Expected opening double quote.");

            var content = string.Empty;

            if (Peek().Kind == QslTokenKind.StringContent)
                content = Advance().Text;

            Consume(
                QslTokenKind.StringEndDouble,
                "Expected closing double quote.");

            return new QslLiteralNode(
                QslLiteralKind.String,
                content);
        }

        private IQslNode ParseStringLiteralSingle()
        {
            Consume(
                QslTokenKind.StringStartSingle,
                "Expected opening single quote.");

            var content = string.Empty;

            if (Peek().Kind == QslTokenKind.StringContent)
                content = Advance().Text;

            Consume(
                QslTokenKind.StringEndSingle,
                "Expected closing single quote.");

            return new QslLiteralNode(
                QslLiteralKind.String,
                content);
        }

        private IQslNode ParseBlock()
        {
            Consume(
                QslTokenKind.LeftBrace,
                "Expected '{' to start block.");

            var statements = new List<IQslNode>();

            while (Peek().Kind != QslTokenKind.RightBrace &&
                   Peek().Kind != QslTokenKind.EndOfFile)
            {
                var statement = ParseExpression();
                statements.Add(statement);

                if (Peek().Kind == QslTokenKind.Semicolon)
                {
                    Advance();
                    continue;
                }

                const string message =
                    "Expected ';' between statements.";

                _errors.Add(message);

                statements.Add(
                    new QslErrorNode(message));

                SynchronizeBlock();

                if (Peek().Kind == QslTokenKind.Semicolon)
                    Advance();
            }

            if (Peek().Kind == QslTokenKind.EndOfFile)
            {
                const string message =
                    "Expected '}' before end of file.";

                _errors.Add(message);

                return new QslBlockNode(statements);
            }

            Consume(
                QslTokenKind.RightBrace,
                "Expected '}' to close block.");

            return new QslBlockNode(statements);
        }

        private IQslNode ParseBooleanLiteral()
        {
            var token = Peek();

            if (token.Kind == QslTokenKind.True)
            {
                Advance();

                return new QslLiteralNode(
                    QslLiteralKind.Boolean,
                    true);
            }

            if (token.Kind == QslTokenKind.False)
            {
                Advance();

                return new QslLiteralNode(
                    QslLiteralKind.Boolean,
                    false);
            }

            var message =
                "Expected boolean literal. " +
                $"Found {token.Kind} '{token.Text}'.";

            _errors.Add(message);
            Advance();

            return new QslErrorNode(message);
        }

        private void SynchronizeCallArguments()
        {
            while (Peek().Kind != QslTokenKind.Comma &&
                   Peek().Kind != QslTokenKind.RightParen &&
                   Peek().Kind != QslTokenKind.EndOfFile)
            {
                Advance();
            }
        }

        private void SynchronizeBlock()
        {
            while (Peek().Kind != QslTokenKind.Semicolon &&
                   Peek().Kind != QslTokenKind.RightBrace &&
                   Peek().Kind != QslTokenKind.EndOfFile)
            {
                Advance();
            }
        }

        private QslToken Peek()
        {
            if (_tokens.Count == 0)
            {
                return new QslToken(
                    QslTokenKind.EndOfFile,
                    string.Empty);
            }

            if (_pos < _tokens.Count)
                return _tokens[_pos];

            return _tokens[_tokens.Count - 1];
        }

        private QslToken Advance()
        {
            var token = Peek();

            if (_pos < _tokens.Count)
                _pos++;

            return token;
        }

        private QslToken Consume(
            QslTokenKind expectedKind,
            string message)
        {
            var token = Peek();

            if (token.Kind == expectedKind)
                return Advance();

            _errors.Add(
                $"{message} Found " +
                $"{token.Kind} '{token.Text}'.");

            return new QslToken(
                QslTokenKind.Unknown,
                token.Text);
        }
    }

using System;
using System.Collections.Generic;
using System.Linq;

namespace QSL.Core.Provenance
{
    public sealed class QslProvenanceBus
    {
        private readonly List<QslProvenanceEntry> _entries = new();
        private readonly HashSet<QslProvenanceEntry> _published = new();

        private readonly Dictionary<string, List<QslProvenanceEntry>> _bySlug
            = new(StringComparer.Ordinal);

        private readonly Dictionary<Guid, List<QslProvenanceEntry>> _byGuid
            = new();

        private readonly Dictionary<QslProvenanceType, List<QslProvenanceEntry>> _byType
            = new();

        private readonly Dictionary<QslProvenanceCategory, List<QslProvenanceEntry>> _byCategory
            = new();

        public event Action<QslProvenanceEntry>? Stream;

        public IDisposable Subscribe(
            Func<QslProvenanceEntry, bool> filter,
            Action<QslProvenanceEntry> handler)
        {
            if (filter is null)
                throw new ArgumentNullException(nameof(filter));

            if (handler is null)
                throw new ArgumentNullException(nameof(handler));

            void SubscriptionHandler(QslProvenanceEntry entry)
            {
                if (filter(entry))
                    handler(entry);
            }

            Stream += SubscriptionHandler;

            return new Subscription(
                () => Stream -= SubscriptionHandler
            );
        }

        internal void Publish(QslProvenanceEntry entry)
        {
            if (entry is null)
                throw new ArgumentNullException(nameof(entry));

            if (!_published.Add(entry))
                return;

            Index(entry);

            Stream?.Invoke(entry);
        }

        private void Index(QslProvenanceEntry entry)
        {
            _entries.Add(entry);

            IndexBySlug(entry);
            IndexByGuid(entry);
            IndexByType(entry);
            IndexByCategory(entry);
        }

        private void IndexBySlug(QslProvenanceEntry entry)
        {
            if (string.IsNullOrWhiteSpace(entry.SubjectSlug))
                return;

            if (!_bySlug.TryGetValue(entry.SubjectSlug, out var entries))
            {
                entries = new List<QslProvenanceEntry>();
                _bySlug[entry.SubjectSlug] = entries;
            }

            entries.Add(entry);
        }

        private void IndexByGuid(QslProvenanceEntry entry)
        {
            if (!_byGuid.TryGetValue(entry.SubjectId, out var entries))
            {
                entries = new List<QslProvenanceEntry>();
                _byGuid[entry.SubjectId] = entries;
            }

            entries.Add(entry);
        }

        private void IndexByType(QslProvenanceEntry entry)
        {
            if (!_byType.TryGetValue(entry.EntryType, out var entries))
            {
                entries = new List<QslProvenanceEntry>();
                _byType[entry.EntryType] = entries;
            }

            entries.Add(entry);
        }

        private void IndexByCategory(QslProvenanceEntry entry)
        {
            if (!_byCategory.TryGetValue(entry.Category, out var entries))
            {
                entries = new List<QslProvenanceEntry>();
                _byCategory[entry.Category] = entries;
            }

            entries.Add(entry);
        }

        public IReadOnlyList<QslProvenanceEntry> GetAll()
        {
            return _entries.ToArray();
        }

        public IReadOnlyList<QslProvenanceEntry> GetForSlug(string slug)
        {
            if (string.IsNullOrWhiteSpace(slug))
                return Array.Empty<QslProvenanceEntry>();

            return _bySlug.TryGetValue(slug, out var entries)
                ? entries.ToArray()
                : Array.Empty<QslProvenanceEntry>();
        }

        public IReadOnlyList<QslProvenanceEntry> GetForGuid(Guid id)
        {
            return _byGuid.TryGetValue(id, out var entries)
                ? entries.ToArray()
                : Array.Empty<QslProvenanceEntry>();
        }

        public IReadOnlyList<QslProvenanceEntry> GetByType(
            QslProvenanceType type)
        {
            return _byType.TryGetValue(type, out var entries)
                ? entries.ToArray()
                : Array.Empty<QslProvenanceEntry>();
        }

        public IReadOnlyList<QslProvenanceEntry> GetByCategory(
            QslProvenanceCategory category)
        {
            return _byCategory.TryGetValue(category, out var entries)
                ? entries.ToArray()
                : Array.Empty<QslProvenanceEntry>();
        }

        public IReadOnlyList<QslProvenanceEntry> GetBetween(
            DateTime start,
            DateTime end)
        {
            if (end < start)
                throw new ArgumentOutOfRangeException(
                    nameof(end),
                    "The end timestamp cannot precede the start timestamp."
                );

            return _entries
                .Where(entry =>
                    entry.TimestampUtc >= start &&
                    entry.TimestampUtc <= end)
                .ToArray();
        }

        public IReadOnlyList<(
            string slug,
            QslProvenanceType type,
            int count)> Compress()
        {
            return _bySlug
                .OrderBy(pair => pair.Key, StringComparer.Ordinal)
                .SelectMany(pair =>
                    pair.Value
                        .GroupBy(entry => entry.EntryType)
                        .OrderBy(group => group.Key)
                        .Select(group => (
                            slug: pair.Key,
                            type: group.Key,
                            count: group.Count()
                        )))
                .ToArray();
        }

        public string Export(
            Func<QslProvenanceEntry, bool>? filter = null)
        {
            IEnumerable<QslProvenanceEntry> entries = _entries;

            if (filter is not null)
                entries = entries.Where(filter);

            return System.Text.Json.JsonSerializer.Serialize(
                entries.ToArray()
            );
        }

        private sealed class Subscription : IDisposable
        {
            private Action? _unsubscribe;

            public Subscription(Action unsubscribe)
            {
                _unsubscribe = unsubscribe;
            }

            public void Dispose()
            {
                var unsubscribe = _unsubscribe;

                if (unsubscribe is null)
                    return;

                _unsubscribe = null;
                unsubscribe();
            }
        }
    }

    {

    public object? Evaluate(IQslNode node)
    {
        var key = node.Kind;

        if (Registry.Evaluators.TryGetValue(key, out var evaluator))
            return evaluator.Evaluate(node, this);

        AddProvenance(
            QslProvenanceType.Error,
            QslProvenanceCategory.Evaluation,

            $"No evaluator registered for node kind: {key}"
        );

        return null;

    {
        if (string.IsNullOrEmpty(idName))
        {
            AddProvenance(
                QslProvenanceType.Error,
                QslProvenanceCategory.Evaluation,
                "QslRuntimeContext",
                "Attempted to resolve a null/empty identifier."
            );

            return null;
        }

        var variable = GetVariable(idName);

        if (variable != null)
            return variable;

        if (Registry.Functions.TryGetValue(idName, out var function))
            return function;

        if (Registry.Descriptors.TryGetValue(idName, out var descriptor))
            return descriptor;

        if (Registry.Capabilities.TryGetValue(idName, out var capability))
            return capability;

        AddProvenance(
            QslProvenanceType.Error,
            QslProvenanceCategory.Evaluation,
            "QslRuntimeContext",
            $"Unresolved identifier: {idName}"
        );

        return null;
    }

using QSL.Abstractions.Base;
using QSL.Abstractions.Runtime;
using QSL.Core.Ast;
using QSL.Core.Base;
using QSL.Core.Lexer;
using QSL.Core.Provenance;
using QSL.Core.Runtime.Operators;

namespace QSL.Core.Runtime.Evaluators
{
    public sealed class QslBinaryEvaluator :
        QslRegistryItem,
        IQslEvaluator
    {
        public override string Type => "evaluator";

        public override string Name => "binary";

        public object? Evaluate(
            IQslNode node,
            QslRuntimeContext context)
        {
            if (node is not QslBinaryNode binary)
            {
                AddProvenance(
                    QslProvenanceType.Error,
                    QslProvenanceCategory.Evaluation,
                    Name,
                    "Binary evaluator received an incompatible node."
                );

                return null;
            }

            var strategy =
                QslKindExtensions.EvaluationStrategy(
                    binary.OperatorKind);

            switch (strategy)
            {
                case QslEvaluationStrategy.Eager:
                    return EvaluateEager(
                        binary,
                        context);

                case QslEvaluationStrategy.ShortCircuitAnd:
                    return EvaluateShortCircuitAnd(
                        binary,
                        context);

                case QslEvaluationStrategy.ShortCircuitOr:
                    return EvaluateShortCircuitOr(
                        binary,
                        context);

                case QslEvaluationStrategy.NullCoalescing:
                    return EvaluateNullCoalescing(
                        binary,
                        context);

                case QslEvaluationStrategy.Arrow:
                    return EvaluateArrow(
                        binary,
                        context);
                
                case QslEvaluationStrategy.Assignment:
                    return EvaluateAssignment(
                        binary,
                        context);
                
                default:
                    AddProvenance(
                        QslProvenanceType.Error,
                        QslProvenanceCategory.Evaluation,
                        Name,
                        $"Unsupported evaluation strategy " +
                        $"'{strategy}' for " +
                        $"'{binary.OperatorKind}'."
                    );

                    return null;
            }
        }

        private object? EvaluateEager(
            QslBinaryNode binary,
            QslRuntimeContext context)
        {
            var leftValue =
                context.Evaluate(binary.Left);

            var rightValue =
                context.Evaluate(binary.Right);

            var executor =
                QslOperatorResolver.Resolve(
                    binary.OperatorKind);

            return executor.ApplyBinary(
                leftValue,
                rightValue,
                context);
        }

        private object? EvaluateShortCircuitAnd(
            QslBinaryNode binary,
            QslRuntimeContext context)
        {
            var leftValue =
                context.Evaluate(binary.Left);

            if (leftValue is not bool leftBoolean)
            {
                AddProvenance(
                    QslProvenanceType.Error,
                    QslProvenanceCategory.Evaluation,
                    Name,
                    "Left operand of '&&' must be boolean."
                );

                return null;
            }

            if (!leftBoolean)
                return false;

            var rightValue =
                context.Evaluate(binary.Right);

            if (rightValue is not bool rightBoolean)
            {
                AddProvenance(
                    QslProvenanceType.Error,
                    QslProvenanceCategory.Evaluation,
                    Name,
                    "Right operand of '&&' must be boolean."
                );

                return null;
            }

            var executor =
                QslOperatorResolver.Resolve(
                    binary.OperatorKind);

            return executor.ApplyBinary(
                leftBoolean,
                rightBoolean,
                context);
        }

        private object? EvaluateShortCircuitOr(
            QslBinaryNode binary,
            QslRuntimeContext context)
        {
            var leftValue =
                context.Evaluate(binary.Left);

            if (leftValue is not bool leftBoolean)
            {
                AddProvenance(
                    QslProvenanceType.Error,
                    QslProvenanceCategory.Evaluation,
                    Name,
                    "Left operand of '||' must be boolean."
                );

                return null;
            }

            if (leftBoolean)
                return true;

            var rightValue =
                context.Evaluate(binary.Right);

            if (rightValue is not bool rightBoolean)
            {
                AddProvenance(
                    QslProvenanceType.Error,
                    QslProvenanceCategory.Evaluation,
                    Name,
                    "Right operand of '||' must be boolean."
                );

                return null;
            }

            var executor =
                QslOperatorResolver.Resolve(
                    binary.OperatorKind);

            return executor.ApplyBinary(
                leftBoolean,
                rightBoolean,
                context);
        }

        private object? EvaluateNullCoalescing(
            QslBinaryNode binary,
            QslRuntimeContext context)
        {
            var leftValue =
                context.Evaluate(binary.Left);

            if (leftValue is not null)
                return leftValue;

            var rightValue =
                context.Evaluate(binary.Right);

            var executor =
                QslOperatorResolver.Resolve(
                    binary.OperatorKind);

            return executor.ApplyBinary(
                leftValue,
                rightValue,
                context);
        }

        private object? EvaluateAssignment(
            QslBinaryNode binary,
            QslRuntimeContext context)
        {
            if (binary.Left is not QslIdentifierNode identifier)
            {
                AddProvenance(
                    QslProvenanceType.Error,
                    QslProvenanceCategory.Evaluation,
                    Name,
                    "Left operand of assignment must be an identifier."
                );

                return null;
            }

            var value =
                context.Evaluate(binary.Right);

            context.SetVariable(
                identifier.Name,
                value);

            return value;
        }
        private object? EvaluateArrow(
            QslBinaryNode binary,
            QslRuntimeContext context)
        {
            context.Evaluate(binary.Left);

            return context.Evaluate(binary.Right);
        }
    }

namespace QSL.Core.Runtime.Evaluators

    public sealed class QslMemberAccessEvaluator :
        QslRegistryItem,
        IQslEvaluator
    {
        public override string Type => "evaluator";

        public override string Name => "member-access";

        public object? Evaluate(
            IQslNode node,
            QslRuntimeContext context)
        {
            if (node is not QslMemberAccessNode memberNode)
            {
                AddProvenance(
                    QslProvenanceType.Error,
                    QslProvenanceCategory.Evaluation,
                    Name,
                    "Member-access evaluator received an incompatible node."
                );

                return null;
            }

            var targetValue = context.Evaluate(memberNode.Target);

            if (targetValue is null)
            {
                Report(
                    memberNode,
                    "Target evaluated to null."
                );

                return null;
            }

            // Capability member access
            if (targetValue is IQslCapability capability)
            {
                var method = capability
                    .GetType()
                    .GetMethod(memberNode.MemberName);

                if (method is not null)
                    return method;

                Report(
                    memberNode,
                    $"Capability '{capability.Name}' has no " +
                    $"member '{memberNode.MemberName}'."
                );

                return null;
            }

            // Descriptor member access
            if (targetValue is IQslDescriptor descriptor)
            {
                descriptor.Apply(context);

                var resolved = context.ResolveIdentifier(
                    memberNode.MemberName
                );

                if (resolved is not null)
                    return resolved;

                Report(
                    memberNode,
                    $"Member '{memberNode.MemberName}' not found " +
                    $"on descriptor '{descriptor.Name}'."
                );

                return null;
            }

            // Plain-object property access
            var targetType = targetValue.GetType();

            var property = targetType.GetProperty(
                memberNode.MemberName
            );

            if (property is not null)
            {
                return property.GetValue(
                    targetValue,
                    null
                );
            }

            // Plain-object field access
            var field = targetType.GetField(
                memberNode.MemberName
            );

            if (field is not null)
                return field.GetValue(targetValue);

            Report(
                memberNode,
                $"Member '{memberNode.MemberName}' not found " +
                $"on object of type '{targetType.Name}'."
            );

            return null;
        }

        private void Report(
            QslMemberAccessNode memberNode,
            string message)
        {
            AddProvenance(
                QslProvenanceType.Error,
                QslProvenanceCategory.Evaluation,
                Name,
                $"{message} " +
                $"(target: '{memberNode.Target}', " +
                $"member: '{memberNode.MemberName}')"
            );
        }
    }

using System;
using System.Collections.Generic;
using QSL.Abstractions.Runtime;
using QSL.Core.Lexer;

namespace QSL.Core.Runtime.Operators
{
    public static class QslOperatorResolver
    {
        private static readonly Dictionary<
            QslTokenKind,
            IQslOperator> Instances =
                new Dictionary<QslTokenKind, IQslOperator>();

        public static IQslOperator Resolve(
            QslTokenKind operatorKind)
        {
            IQslOperator existing;

            if (Instances.TryGetValue(
                    operatorKind,
                    out existing))
            {
                return existing;
            }

            var executorType =
                QslKindExtensions.ExecutorType(
                    operatorKind);

            if (!typeof(IQslOperator).IsAssignableFrom(
                    executorType))
            {
                throw new InvalidOperationException(
                    $"Executor type '{executorType.FullName}' " +
                    $"for QslTokenKind.{operatorKind} does not " +
                    $"implement {nameof(IQslOperator)}.");
            }

            var constructor =
                executorType.GetConstructor(Type.EmptyTypes);

            if (constructor == null)
            {
                throw new InvalidOperationException(
                    $"Executor type '{executorType.FullName}' " +
                    $"for QslTokenKind.{operatorKind} must have " +
                    "a public parameterless constructor.");
            }

            var instance =
                Activator.CreateInstance(executorType)
                as IQslOperator;

            if (instance == null)
            {
                throw new InvalidOperationException(
                    $"Unable to create executor " +
                    $"'{executorType.FullName}' for " +
                    $"QslTokenKind.{operatorKind}.");
            }

            Instances.Add(
                operatorKind,
                instance);

            return instance;
        }

        public static bool IsResolved(
            QslTokenKind operatorKind)
        {
            return Instances.ContainsKey(
                operatorKind);
        }

        public static void Clear()
        {
            Instances.Clear();
        }
    }
}

        public void RegisterType(IQslObject obj)
        {
            if (obj == null)
                throw new ArgumentNullException(nameof(obj));

            if (obj is QslRuntimeObject runtimeObject)
            {
                runtimeObject.AppendSlug(
                    $"{runtimeObject.Type}-{runtimeObject.Name}");
            }

            switch (obj)
            {
                case IQslFunction function:
                    Register(

                        function.Name,
                        function,
                        "function");
                    return;

                case IQslCapability capability:
                    Register(
                        _capabilities,

                        capability,

                    return;

                    return;

                case IQslOperator op:

                        op.Name,
                        op,
                        "operator");
                    return;

                    AddProvenance(
                        QslProvenanceType.Error,
                        QslProvenanceCategory.Evaluation,

                        blockFactory.Name,
                        blockFactory,
                        "block-factory");

                        QslProvenanceType.Error,
                        QslProvenanceCategory.Evaluation,

                runtimeObject.AppendSlug(

                    QslProvenanceType.Error,
                    QslProvenanceCategory.Evaluation,
                    nameof(QslRuntimeRegistry),
                    $"Evaluator already registered for node kind: {kind}"

            _evaluators.Add(

                );

                return null;
            }

                nameof(QslRuntimeRegistry),
                $"No evaluator registered for node kind: {node.Kind}"

            T item,

                    QslProvenanceCategory.Evaluation,
                    nameof(QslRuntimeRegistry),
                    $"Cannot register {itemType} with an empty name."
                );

                return;
            }

            if (registry.ContainsKey(key))
            {
                AddProvenance(
                    QslProvenanceType.Error,
                    QslProvenanceCategory.Evaluation,
                    nameof(QslRuntimeRegistry),
                    $"{itemType} '{key}' is already registered."
                );

                return;
            }

            registry.Add(
                key,
                item);
        }

        {
            try
            {
                var parseResult = Parse(source);

                if (parseResult.HasDiagnostics)
                {
                    return QslProcessingResult.ParseFailure(
                        parseResult.Ast,
                        parseResult.Diagnostics
                    );
                }

                var value = _context.Evaluate(
                    parseResult.Ast
                );

                return QslProcessingResult.Success(
                    parseResult.Ast,
                    value
                );
            }
            catch (Exception exception)
            {
                return QslProcessingResult.ExceptionFailure(
                    exception
                );
            }
        }
    }

    {
        public IQslNode Ast { get; }

        public IReadOnlyList<string> Diagnostics { get; }

        public bool HasDiagnostics =>
            Diagnostics.Count > 0;

        public QslParseResult(
            IQslNode ast,
            IReadOnlyList<string> diagnostics)
        {
            Ast = ast;
            Diagnostics = diagnostics;
        }
    }

    public sealed class QslProcessingResult
    {
        public IQslNode? Ast { get; }

        public object? Value { get; }

        public IReadOnlyList<string> ParserDiagnostics { get; }

        public Exception? Exception { get; }

        public bool ParseFailed =>
            ParserDiagnostics.Count > 0;

        public bool HasException =>
            Exception is not null;

        public bool Succeeded =>
            !ParseFailed &&
            !HasException;

        private QslProcessingResult(
            IQslNode? ast,
            object? value,
            IReadOnlyList<string> parserDiagnostics,
            Exception? exception)
        {
            Ast = ast;
            Value = value;
            ParserDiagnostics = parserDiagnostics;
            Exception = exception;
        }

        public static QslProcessingResult Success(
            IQslNode ast,
            object? value)
        {
            return new QslProcessingResult(
                ast,
                value,
                Array.Empty<string>(),
                null
            );
        }

        public static QslProcessingResult ParseFailure(
            IQslNode ast,
            IReadOnlyList<string> diagnostics)
        {
            return new QslProcessingResult(
                ast,
                null,
                diagnostics,
                null
            );
        }

        public static QslProcessingResult ExceptionFailure(
            Exception exception)
        {
            return new QslProcessingResult(
                null,
                null,
                Array.Empty<string>(),
                exception
            );
        }
